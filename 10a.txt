# 6. Go to the Fabric test network
cd ~/fabric-samples/test-network

# 7. Start the network
./network.sh up

# 8. Create mychannel
./network.sh createChannel -c mychannel

# 9. Create carchannel
./network.sh createChannel -c carchannel

# 10. Deploy the modified FabCar chaincode
./network.sh deployCC \
  -ccn fabcar \
  -ccp ~/fabric-samples-1.4/chaincode/fabcar/javascript \
  -ccl javascript \
  -c mychannel

# 11. Query cars by owner
# Run this only after successful deployment.
